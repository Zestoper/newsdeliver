## 1. 타입 힌트 (Type Hints)

```python
# 기본 타입
def get_user(user_id: str) -> dict | None:  # Python 3.10+: Union 대신 |
    ...

# 컨테이너 타입
def fetch_news() -> list[dict]:             # list of dict
    ...

def classify(scores: dict[str, int]) -> str | None:  # dict with typed keys/values
    ...

# Optional (None 가능)
def add_comment(parent_id: int | None = Form(None)):  # None이 기본값
    ...
```

**왜 중요한가?**
- FastAPI는 타입 힌트를 기반으로 자동 요청 파싱 및 Swagger 문서를 생성합니다.
- 런타임 오류를 미리 잡는 데 도움을 주고, IDE 자동완성이 작동합니다.

## 2. async / await

```python
# FastAPI 라우터는 async def 또는 일반 def 둘 다 가능
@router.get("/news")
async def news_list(request: Request):          # I/O 작업이 있으면 async 권장
    async with httpx.AsyncClient() as client:  # 비동기 HTTP 클라이언트
        r = await client.get(url)              # await으로 실행 완료 대기
    return r.json()

# APScheduler에 등록하는 백그라운드 작업은 일반 함수로
def _newsletter_job():                         # async 아님 (별도 스레드에서 실행)
    run_newsletter_job(base_url=...)
```

**async vs sync 선택 기준:**
- `async def` : 데이터베이스, 외부 API 등 I/O 대기가 있는 경우
- 일반 `def` : CPU 집약적 작업, 스레드 풀에서 실행되는 백그라운드 작업
- FastAPI는 일반 `def` 라우터를 자동으로 스레드 풀에서 실행합니다.

## 3. asynccontextmanager

```python
from contextlib import asynccontextmanager

@asynccontextmanager
async def lifespan(app: FastAPI):
    # 서버 시작 시 실행
    scheduler.start()
    yield              # ← 서버 실행 중 (여기서 멈춤)
    # 서버 종료 시 실행
    scheduler.shutdown()

app = FastAPI(lifespan=lifespan)
```

**일반 with문의 async 버전:**
```python
with engine.connect() as conn:        # 동기 컨텍스트 매니저
    ...

async with httpx.AsyncClient() as client:  # 비동기 컨텍스트 매니저
    ...
```

## 4. SQLAlchemy text() 사용법

```python
from sqlalchemy import text

with engine.connect() as conn:
    # Named parameter (SQL injection 방어)
    row = conn.execute(
        text("SELECT * FROM news WHERE id = :id AND status = :status"),
        {"id": news_id, "status": "published"}   # 딕셔너리로 파라미터 전달
    ).fetchone()
    
    # 결과 조회 방법들
    row = result.fetchone()             # 첫 번째 행
    rows = result.fetchall()            # 모든 행 (list)
    value = result.scalar()             # 단일 값 (SELECT COUNT(*) 등)
    
    # Row → dict 변환
    data = dict(row._mapping)           # SQLAlchemy 2.x 방식
    
    # 쓰기 작업 후 반드시 commit
    conn.execute(text("INSERT INTO ..."), {...})
    conn.commit()
```

## 6. 데코레이터 (@)

```python
# FastAPI 라우터 등록
@router.get("/path")           # GET 요청
@router.post("/path")          # POST 요청
@router.put("/path/{id}")      # PUT 요청 (경로 파라미터 포함)
@router.delete("/path/{id}")   # DELETE 요청
@router.websocket("/ws/{id}")  # WebSocket

# 함수 데코레이터 동작 원리
def my_decorator(func):
    def wrapper(*args, **kwargs):
        print("before")
        result = func(*args, **kwargs)
        print("after")
        return result
    return wrapper

@my_decorator
def hello():   # hello = my_decorator(hello)
    print("hello")
```

```python
    SELECT n.*, c.name AS category_name
```

## 14. httpx (HTTP 클라이언트)

```python
import httpx

# 동기
with httpx.Client(timeout=10, follow_redirects=True) as client:
    res = client.get(url, headers={"Authorization": f"Bearer {token}"})
    if res.status_code == 200:
        data = res.json()

# 비동기
async with httpx.AsyncClient(timeout=5) as client:
    r = await client.get(url)
    
# POST (JSON)
res = httpx.post(
    url,
    headers={"Authorization": f"Bearer {api_key}"},
    json={"model": "llama-3.1-8b-instant", "messages": [...]},
    timeout=15
)
res.raise_for_status()  # 4xx/5xx 시 예외 발생
```

## 15. 정규표현식 (re 모듈)

```python
import re

# 태그 제거
clean = re.sub(r'<[^>]+>', '', html)           # HTML 태그 제거
clean = re.sub(r'<(script|style)[^>]*>[\s\S]*?</(script|style)>', '', html, flags=re.IGNORECASE)

# 매칭
match = re.search(r'<meta[^>]+property=["\']og:image["\'][^>]+content=["\']([^"\']+)["\']', html)
if match:
    image_url = match.group(1)   # 캡처 그룹 1번

# 단어 경계 매칭
re.search(r'\b' + re.escape(keyword) + r'\b', text)  # 'tech'가 'technology' 안에서 매칭 안 됨

# 플래그
re.IGNORECASE    # 대소문자 무시
re.MULTILINE     # ^, $가 줄 시작/끝에 매칭
re.DOTALL        # .이 줄바꿈도 매칭

# 컴파일된 패턴 (반복 사용 시 성능 향상)
_PATTERN = re.compile(r'...', re.MULTILINE | re.IGNORECASE)
_PATTERN.sub('', text)
```
